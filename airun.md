WITH __ghosts AS (
  SELECT
    ghost_id,
    ghost_name,
    ghost_type,
    status,
    threat_level,
    confidence_score
  FROM ghost_detection.app.ghosts
)
SELECT
  ghost_id,
  ghost_name,
  ghost_type,
  status,
  threat_level,
  confidence_score,
  COUNT(*) OVER () AS total_ghosts,
  MIN(threat_level) OVER () AS min_threat_level,
  MAX(threat_level) OVER () AS max_threat_level,
  COUNT(DISTINCT threat_level) OVER () AS distinct_threat_levels
FROM __ghosts
ORDER BY
  CASE
    WHEN threat_level = 'Extreme'
    THEN 1
    WHEN threat_level = 'High'
    THEN 2
    WHEN threat_level = 'Medium'
    THEN 3
    WHEN threat_level = 'Low'
    THEN 4
    ELSE 5
  END,
  ghost_name

WITH __ghost_sightings AS (
  SELECT
    ghost_id,
    location_name,
    sighting_datetime,
    sighting_id,
    verified,
    witness_name,
    emf_reading,
    paranormal_activity_level,
    temperature_celsius
  FROM ghost_detection.app.ghost_sightings
)
SELECT
  gs.sighting_id,
  gs.ghost_id,
  gs.location_name,
  gs.witness_name,
  gs.sighting_datetime,
  gs.paranormal_activity_level,
  gs.emf_reading,
  gs.temperature_celsius,
  gs.verified,
  COUNT(*) OVER () AS total_sightings,
  COUNT(DISTINCT gs.location_name) OVER () AS unique_locations,
  MIN(gs.paranormal_activity_level) OVER () AS min_paranormal_activity_level,
  MAX(gs.paranormal_activity_level) OVER () AS max_paranormal_activity_level,
  AVG(gs.paranormal_activity_level) OVER () AS avg_paranormal_activity_level
FROM __ghost_sightings AS gs
WHERE
  UPPER(gs.location_name) LIKE '%NEW YORK%'
ORDER BY
  gs.paranormal_activity_level DESC NULLS LAST,
  gs.sighting_datetime DESC NULLS LAST
